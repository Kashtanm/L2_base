function whatCentury(year) {
  const c = Math.ceil(Number(year) / 100);
  const last2 = c % 100;
  let suffix = 'th';
  if (last2 < 11 || last2 > 13) {
    suffix = { 1: 'st', 2: 'nd', 3: 'rd' }[c % 10] || 'th';
  }
  return c + suffix;
}
