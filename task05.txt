function primeFactors(n) {
  let result = "";
  let p = 2;

  while (p * p <= n) {
    let count = 0;

    while (n % p === 0) {
      n = n / p;
      count++;
    }

    if (count > 0) {
      result += "(" + p;
      if (count > 1) {
        result += "**" + count;
      }
      result += ")";
    }

    p++;
  }

  if (n > 1) {
    result += "(" + n + ")";
  }

  return result;
}
